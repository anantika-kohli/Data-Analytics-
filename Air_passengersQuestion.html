#(a)
library(astsa)
## Warning: package 'astsa' was built under R version 4.4.3
data(oil)
data(gas)

ts.plot(oil,gas,col=c("blue","red"),lwd=2, xlab="Time in weeks", ylab="Price", main="Weekly Oil (blue) and Gas (red) Prices")
legend("topleft", legend=c("Oil (dollars per barrel)", "Gas (cents per gallon)"), col=c("blue","red"), lwd=2)

#(c)
log_of_oil=diff(log(oil))
log_of_gas=diff(log(gas))
ts.plot(log_of_oil, log_of_gas, col=c("blue","red"), lwd=1, xlab="Time in weeks", ylab="Tranformed price data", main="Weekley returns of Oil and Gas")
legend("topright", legend=c("Oil", "Gas"), col=c("blue", "red"), lwd=2)

#(c) ACF plots
acf(log_of_oil, lag.max=40, main = "ACF of transformed Oil series", xlab="Lag (weeks)")

acf(log_of_gas, lag.max=40, main = "ACF of transformed Gas series", xlab="Lag (weeks)")

#(d)
ccf(log_of_oil, log_of_gas, lag.max=40, main ="CCF of Transformed Oil and Gas Series", xlab="Lag (weeks)", ylab = "Cross correlation")

#(e)
# oil prices are ahead of gas
oil_lead1=lag(log_of_oil, -1)
oil_lead2=lag(log_of_oil, -2)
oil_lead3=lag(log_of_oil, -3)


#1st week lead
plot(oil_lead1, log_of_gas, main="Oil leads by 1 week")
lines(lowess(oil_lead1, log_of_gas), col="blue", lwd=2)

#2nd week lead
plot(oil_lead2, log_of_gas, main="Oil leads by 2 weeks")
lines(lowess(oil_lead2, log_of_gas), col="blue", lwd=2)

#3rd week lead
plot(oil_lead3, log_of_gas, main="Oil leads by 3 weeks")
lines(lowess(oil_lead3, log_of_gas), col="blue", lwd=2)
